-- ── TARGET PICKER ─────────────────────────────────────────────────────────────


local wiredHumanoids = {}


local function getNearestTarget()
    if (tick() - lastKillTime) < KILL_COOLDOWN then return nil end


    local nearest, nearestDist = nil, math.huge
    local selfChar = player.Character
    local selfRoot = selfChar and selfChar:FindFirstChild("HumanoidRootPart")
    if not selfRoot then return nil end


    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and not isSameTeam(p) and p.Character then
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            local hum  = p.Character:FindFirstChild("Humanoid")
            if root and hum and hum.Health > 0 then
                local dist = (root.Position - selfRoot.Position).Magnitude
                if dist < nearestDist then
                    nearestDist = dist
                    nearest = { root = root, head = p.Character:FindFirstChild("Head") or root }
                end
            end
        end
    end


    for _, model in ipairs(workspace:GetDescendants()) do
        if model:IsA("Model") and model ~= selfChar then
            local hum  = model:FindFirstChildOfClass("Humanoid")
            local root = model:FindFirstChild("HumanoidRootPart")
            if hum and root and hum.Health > 0 then
                local isPlayer = false
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character == model then isPlayer = true; break end
                end
                if not isPlayer then
                    local dist = (root.Position - selfRoot.Position).Magnitude
                    if dist < nearestDist then
                        nearestDist = dist
                        nearest = { root = root, head = model:FindFirstChild("Head") or root }
                    end
                end
            end
        end
    end


    if nearest then
        local hum = nearest.root.Parent:FindFirstChildOfClass("Humanoid")
        if hum and not wiredHumanoids[hum] then
            wiredHumanoids[hum] = true
            hum.Died:Connect(function()
                onTargetKilled()
                wiredHumanoids[hum] = nil
            end)
        end
    end


    return nearest
end


-- ── NIL HELPERS ───────────────────────────────────────────────────────────────


local function getNilHead(targetHead)
    if not getnilinstances then return targetHead end
    local debugId = targetHead:GetDebugId()
    for _, obj in ipairs(getnilinstances()) do
        if obj.Name == "Head" and obj:GetDebugId() == debugId then return obj end
    end
    return targetHead
end


local function getNilByName(name)
    if not getnilinstances then return nil end
    for _, obj in ipairs(getnilinstances()) do
        if obj.Name == name and obj:IsA("RemoteEvent") then return obj end
    end
    return nil
end


-- ── AIM VARIANCE ──────────────────────────────────────────────────────────────


local AIM_VARIANCE_MIN = 0.15
local AIM_VARIANCE_MAX = 0.85


local function getAimVariance(dist)
    local scale  = math.clamp(dist / 60, 0.5, 1.4)
    local radius = (AIM_VARIANCE_MIN + math.random() * (AIM_VARIANCE_MAX - AIM_VARIANCE_MIN)) * scale
    local angle  = math.random() * math.pi * 2
    return Vector3.new(
        math.cos(angle) * radius,
        math.sin(angle) * radius * 0.45,
        0
    )
end


-- ── SHOTGUN PELLET BUILDER ────────────────────────────────────────────────────


local function buildPellets(data, hitCount, bodyRatio, clientHitSizeHead, clientHitSizeBody)
    local hits   = {}
    local SPREAD = 0.4
    for i = 1, hitCount do
        local isBody   = math.random() < bodyRatio
        local instance = isBody and (data.targetRoot or data.targetHead) or data.nilHead
        local offset   = Vector3.new(
            (math.random()-0.5)*SPREAD,
            (math.random()-0.5)*SPREAD,
            (math.random()-0.5)*SPREAD)
        hits[i] = {
            Normal        = data.mouseNormal,
            ClientHitSize = isBody
                and (clientHitSizeBody or Vector3.new(2,2,1))
                or  (clientHitSizeHead or Vector3.new(1,1,1)),
            Position      = data.targetPos + offset,
            Instance      = instance,
            Distance      = data.dist,
            StartPos      = data.startPos,
            Material      = Enum.Material.Plastic,
        }
    end
    return hits
end

-- ── GUN LOOKUP ────────────────────────────────────────────────────────────────


function getEquippedGun(selfChar)
    local function matchRegistry(item)
        local lower = item.Name:lower()
        for _, entry in ipairs(GUN_REGISTRY) do
            if lower == entry.name:lower() then return item, entry end
        end
        return nil, nil
    end
    for _, item in ipairs(selfChar:GetChildren()) do
        if item:IsA("Tool") then
            local gun, entry = matchRegistry(item)
            if gun then return gun, entry end
        end
    end
    for _, item in ipairs(player.Backpack:GetChildren()) do
        if item:IsA("Tool") then
            local gun, entry = matchRegistry(item)
            if gun then return gun, entry end
        end
    end
    return nil, nil
end


-- ── FIRE SEQUENCE ─────────────────────────────────────────────────────────────


function fireAtTarget()
    local selfChar = player.Character
    if not selfChar then return end
    local selfRoot = selfChar:FindFirstChild("HumanoidRootPart")
    if not selfRoot then return end


    local target = getNearestTarget()
    if not target then return end


    local gun, entry = getEquippedGun(selfChar)
    if not gun then return end


    local targetHead = target.head
    local targetRoot = target.root
    local targetPos  = targetHead.Position
    local startPos   = selfRoot.Position
    local dist       = (targetPos - startPos).Magnitude


    local effectiveTargetPos = targetPos
    if not entry.isShotgun then
        effectiveTargetPos = targetPos + getAimVariance(dist)
    end


    local dir        = effectiveTargetPos - startPos
    local baseNormal = dir.Magnitude > 0 and dir.Unit or Vector3.new(0, 0, -1)


    local JITTER = 0.015
    local function driftedNormal()
        return (baseNormal + Vector3.new(
            (math.random()-0.5)*JITTER,
            (math.random()-0.5)*JITTER,
            (math.random()-0.5)*JITTER
        )).Unit
    end


    local vel     = selfRoot.AssemblyLinearVelocity
    local moveDir = vel.Magnitude > 0.1 and vel.Unit or Vector3.new(0, 0, 0)


    local holdTime = 0.09 + math.random() * 0.02
    local fireHold = 0.08 + math.random() * 0.02


    if not entry.skipEBind then
        Abilities:FireServer({ MoveDirection=moveDir, Bind="E", PressType="Press",
            HoldTime=0, MouseNormal=driftedNormal(),
            MouseHit=targetHead, MousePosition=effectiveTargetPos })
        task.wait(holdTime)
        Abilities:FireServer({ MoveDirection=moveDir, Bind="E", PressType="Release",
            HoldTime=holdTime, MouseNormal=driftedNormal(),
            MouseHit=targetHead, MousePosition=effectiveTargetPos })
        task.wait(0.01)
    end


    if not entry.skipFireBind then
        Abilities:FireServer({ MoveDirection=moveDir, Bind="FIRE", PressType="Press",
            HoldTime=0, MouseNormal=driftedNormal(),
            MouseHit=targetHead, MousePosition=effectiveTargetPos })
        task.wait(fireHold)
        Abilities:FireServer({ MoveDirection=moveDir, Bind="FIRE", PressType="Release",
            HoldTime=fireHold, MouseNormal=driftedNormal(),
            MouseHit=targetHead, MousePosition=effectiveTargetPos })
        task.wait(0.01)
    end


    entry.fireFunc(gun, {
        targetHead  = targetHead,
        targetRoot  = targetRoot,
        targetPos   = effectiveTargetPos,
        startPos    = startPos,
        dist        = dist,
        mouseNormal = driftedNormal(),
        nilHead     = getNilHead(targetHead),
    })
end
